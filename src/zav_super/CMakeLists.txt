cmake_minimum_required(VERSION 3.16)
project(zav_super)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp nav_msgs geometry_msgs std_srvs std_msgs px4_msgs mars_quadrotor_msgs ommpc_core)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(ommpc_px4 src/ommpc_px4.cpp)
target_compile_features(ommpc_px4 PUBLIC cxx_std_17)
target_include_directories(ommpc_px4 PRIVATE include)
target_link_libraries(ommpc_px4 ommpc_core::ommpc_core)
ament_target_dependencies(ommpc_px4 rclcpp nav_msgs geometry_msgs std_srvs std_msgs px4_msgs mars_quadrotor_msgs)
install(TARGETS ommpc_px4 DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS src/lio_adapter.py src/benchmark_mission.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch rviz meshes DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(control_math_test test/control_math.cpp)
  target_compile_features(control_math_test PUBLIC cxx_std_17)
  target_include_directories(control_math_test PRIVATE include)
  target_link_libraries(control_math_test ommpc_core::ommpc_core)
  ament_target_dependencies(control_math_test mars_quadrotor_msgs)
  add_test(NAME control_math COMMAND control_math_test)
endif()
ament_package()
